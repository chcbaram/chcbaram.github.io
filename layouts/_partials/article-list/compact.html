{{- /* 테마의 같은 이름 partial 을 덮어쓴다. 바꾼 것은 두 가지. 1. taxonomy term(프로젝트 · 태그 · 시리즈)이면 제목 옆에 글 수를 붙인다. 테마 원본은 description 이 없을 때만 "N pages" 를 보여주는데, 우리 프로젝트 term 에는 설명이 있어서 글 수가 영영 안 보인다. 2. 프로젝트 term 중 최근에 글이 올라온 것에 NEW 배지를 붙인다. term 의 .Date 는 그 안에서 가장 최근 글의 날짜다. 기준 일수는 hugo.toml 의 params.recentDays (없으면 14일). 정적 사이트라 빌드 시점 기준이다 — 푸시할 때마다 다시 계산된다. 3. 아카이브 목록의 글에도 제목 위에 프로젝트 이름을 붙인다 — 홈과 같은 자리다. 여기서는 항목 전체가 라서 링크를 겹칠 수 없으므로 span 으로 넣는다. */ -}} {{- $IsTerm := in .Kind "term" -}} {{- $isNew := false -}} {{- if and $IsTerm (eq .Section "projects") (not .Date.IsZero) -}} {{- $days := site.Params.recentDays | default 14 -}} {{- $age := div (sub now.Unix .Date.Unix) 86400 -}} {{- $isNew = lt $age $days -}} {{- end -}}
{{- if not $IsTerm -}} {{- with .GetTerms "projects" -}} {{- range . }}{{ .LinkTitle }}{{ end -}} {{- end -}} {{- end -}}

{{- .Title -}} {{- if $IsTerm }}{{ len .Pages }}{{ end -}} {{- if $isNew }}{{ T "recentlyUpdated" }}{{ end -}}

{{- $description := .Params.description -}} {{- if $IsTerm -}} {{- $description = $description | default (T "list.page" (len .Pages)) -}} {{- end -}} {{- if or $description (not .Date.IsZero) -}}
{{- if or $IsTerm .Date.IsZero -}} {{- $description -}} {{- else if not .Date.IsZero -}} {{- /* .article-meta 는 flex column 이다. 날짜와 (JS 가 붙이는) 댓글 수를 한 줄에 두려면 테마의 .inline-meta 로 감싸야 한다. */ -}}
{{- .Date | time.Format .Site.Params.dateFormat.published -}}
{{- end }}
{{- end -}}
